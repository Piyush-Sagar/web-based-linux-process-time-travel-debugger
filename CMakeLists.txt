cmake_minimum_required(VERSION 3.20)
project(time_travel_debugger LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  message(FATAL_ERROR "Backend requires x86-64 Linux. On Windows build inside Ubuntu WSL2.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(nlohmann_json 3.10 REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
foreach(demo crash clean forbidden loop)
  add_executable(demo_${demo} demos/${demo}.cpp)
  target_compile_options(demo_${demo} PRIVATE -O0 -g -fno-pie -fno-stack-protector -fno-exceptions -fno-rtti -mno-sse -mno-mmx -msoft-float -Wall -Wextra)
  target_link_options(demo_${demo} PRIVATE -nostdlib -static -no-pie -Wl,-e,_start -Wl,--build-id=none)
endforeach()
add_library(engine backend/engine.cpp)
target_include_directories(engine PUBLIC backend)
target_link_libraries(engine PUBLIC nlohmann_json::nlohmann_json)
target_compile_definitions(engine PUBLIC DEMO_DIR="${CMAKE_BINARY_DIR}")
target_compile_options(engine PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ttd backend/cli.cpp)
target_link_libraries(ttd PRIVATE engine pthread)
add_executable(ttd-server backend/server.cpp)
target_link_libraries(ttd-server PRIVATE engine Boost::system pthread)
target_compile_definitions(ttd-server PRIVATE FRONTEND_DIR="${CMAKE_SOURCE_DIR}/frontend/dist")
target_compile_options(ttd-server PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/demos.json"
  COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/scripts/demo_manifest.py" "${CMAKE_BINARY_DIR}/demos.json" $<TARGET_FILE:demo_crash> $<TARGET_FILE:demo_clean> $<TARGET_FILE:demo_forbidden> $<TARGET_FILE:demo_loop>
  DEPENDS demo_crash demo_clean demo_forbidden demo_loop "${CMAKE_SOURCE_DIR}/scripts/demo_manifest.py"
  VERBATIM)
add_custom_target(demo_manifest ALL DEPENDS "${CMAKE_BINARY_DIR}/demos.json")
add_dependencies(engine demo_manifest)
add_test(NAME engine_replay COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/engine_test.py $<TARGET_FILE:ttd>)
add_test(NAME api_sessions COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/api_test.py)
set_tests_properties(engine_replay api_sessions PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
